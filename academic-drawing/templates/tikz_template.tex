\newcommand{\icn}[5]{%
  \begin{tabular}{@{}l@{\hspace{1.6mm}}l@{}}%
    \raisebox{-0.25ex}{\tikz[baseline=(glyph.base)]{\node[icon plate,fill=#5!40,text=#5,font=\fontsize{#1}{#1}\selectfont] (glyph) {#2};}} &
    \begin{tabular}[c]{@{}l@{}}%
      {\sanssemiboldcjk\color{ink}\fontsize{9.8}{12}\selectfont #3}\\[0.08em]%
      {\wenkaicjk\color{muted}\fontsize{7.8}{9.6}\selectfont #4}%
    \end{tabular}%
  \end{tabular}%
}
